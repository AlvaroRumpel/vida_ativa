Create spike-supabase/tests/ directory and two SQL test files. These files are NOT meant to run successfully yet (schema doesn't exist) — they define the test scenarios that WILL be executed after Plans 02 and 03 apply RLS and seed data.

NOTE on JWT syntax: Supabase stores tenant_id in app_metadata. The JWT claims object must be:
`{"sub":"user-uuid","app_metadata":{"tenant_id":"tenant-uuid","role":"client"}}`
The RLS policy accesses it as: `(auth.jwt() ->> 'app_metadata' ->> 'tenant_id')::uuid`

Create spike-supabase/tests/rls_isolation_test.sql with exactly this content:

```sql
-- =============================================================================
-- RLS Isolation Test Suite
-- Run AFTER: 01_schema.sql + 02_rls.sql + 03_seed.sql applied
-- Run with: psql "$SUPABASE_DB_URL" -f spike-supabase/tests/rls_isolation_test.sql
--
-- Seed UUID constants (from 03_seed.sql):
--   Tenant A: 10000000-0000-0000-0000-000000000001
--   Tenant B: 20000000-0000-0000-0000-000000000001
--   Client A (Tenant A): see auth.users after signup
--   Client B (Tenant B): see auth.users after signup
--
-- NOTE: SET LOCAL request.jwt.claims only works inside a transaction block.
-- Each test uses BEGIN/ROLLBACK to simulate a user session.
-- =============================================================================

-- -----------------------------------------------------------------------------
-- Attack 1: Cross-tenant SELECT — client in Tenant A tries to read Tenant B rows
-- Expected: 0 rows returned from Tenant B (RLS blocks access)
-- -----------------------------------------------------------------------------
SELECT '=== Attack 1: Cross-tenant SELECT on bookings ==='::TEXT AS test;

BEGIN;
  SET LOCAL role TO authenticated;
  SET LOCAL request.jwt.claims = '{"sub":"00000000-0000-0000-0000-000000000011","app_metadata":{"tenant_id":"10000000-0000-0000-0000-000000000001","role":"client"}}';

  -- This query should return 0 rows (Tenant B data not visible to Tenant A client)
  SELECT
    CASE
      WHEN COUNT(*) = 0 THEN 'PASS: Cross-tenant SELECT blocked (0 rows from Tenant B)'
      ELSE CONCAT('FAIL: Cross-tenant SELECT LEAKED ', COUNT(*)::TEXT, ' rows from Tenant B')
    END AS result
  FROM bookings
  WHERE tenant_id = '20000000-0000-0000-0000-000000000001'::uuid;

ROLLBACK;

-- -----------------------------------------------------------------------------
-- Attack 2: Cross-tenant SELECT on slots — client in Tenant A reads Tenant B slots
-- Expected: 0 rows returned from Tenant B slots
-- -----------------------------------------------------------------------------
SELECT '=== Attack 2: Cross-tenant SELECT on slots ==='::TEXT AS test;

BEGIN;
  SET LOCAL role TO authenticated;
  SET LOCAL request.jwt.claims = '{"sub":"00000000-0000-0000-0000-000000000011","app_metadata":{"tenant_id":"10000000-0000-0000-0000-000000000001","role":"client"}}';

  SELECT
    CASE
      WHEN COUNT(*) = 0 THEN 'PASS: Cross-tenant SELECT on slots blocked (0 rows from Tenant B)'
      ELSE CONCAT('FAIL: Cross-tenant SELECT on slots LEAKED ', COUNT(*)::TEXT, ' Tenant B slots')
    END AS result
  FROM slots
  WHERE tenant_id = '20000000-0000-0000-0000-000000000001'::uuid;

ROLLBACK;

-- -----------------------------------------------------------------------------
-- Attack 3: Admin in Tenant A sees ALL Tenant A bookings (not just own)
-- Expected: COUNT > 1 (admin sees all tenant rows, not just own user_id)
-- This validates D-07: admin policy is broader than client policy
-- -----------------------------------------------------------------------------
SELECT '=== Attack 3: Admin SELECT — sees all Tenant A bookings ==='::TEXT AS test;

BEGIN;
  SET LOCAL role TO authenticated;
  SET LOCAL request.jwt.claims = '{"sub":"00000000-0000-0000-0000-000000000012","app_metadata":{"tenant_id":"10000000-0000-0000-0000-000000000001","role":"admin"}}';

  SELECT
    CASE
      WHEN COUNT(*) > 0 THEN CONCAT('PASS: Admin sees ', COUNT(*)::TEXT, ' Tenant A bookings (all users)')
      ELSE 'FAIL: Admin sees 0 bookings — admin policy too restrictive'
    END AS result
  FROM bookings
  WHERE tenant_id = '10000000-0000-0000-0000-000000000001'::uuid;

ROLLBACK;

SELECT 'rls_isolation_test.sql complete. Review PASS/FAIL above.'::TEXT AS summary;
```

Create spike-supabase/tests/rls_insert_isolation_test.sql with exactly this content:

```sql
-- =============================================================================
-- RLS WITH CHECK Isolation Test (Pitfall 2 Guard)
-- Run AFTER: 01_schema.sql + 02_rls.sql applied
-- Run with: psql "$SUPABASE_DB_URL" -f spike-supabase/tests/rls_insert_isolation_test.sql
--
-- Tests that INSERT policies have WITH CHECK clause (Pitfall 2 from PITFALLS.md).
-- Without WITH CHECK, users can INSERT into wrong tenant — critical security gap.
-- =============================================================================

-- -----------------------------------------------------------------------------
-- Attack: Client in Tenant A tries to INSERT a booking for Tenant B
-- Expected: INSERT fails with "violates row-level security policy" error
-- -----------------------------------------------------------------------------
SELECT '=== WITH CHECK Test: Cross-tenant INSERT attempt ==='::TEXT AS test;

BEGIN;
  SET LOCAL role TO authenticated;
  SET LOCAL request.jwt.claims = '{"sub":"00000000-0000-0000-0000-000000000011","app_metadata":{"tenant_id":"10000000-0000-0000-0000-000000000001","role":"client"}}';

  -- Attempt to INSERT a booking into Tenant B (should be blocked by WITH CHECK)
  -- This will raise an exception if RLS WITH CHECK is correctly configured
  DO $$
  BEGIN
    INSERT INTO bookings (id, tenant_id, user_id, slot_id, booking_date, status)
    VALUES (
      gen_random_uuid(),
      '20000000-0000-0000-0000-000000000001'::uuid,  -- Tenant B (attacker's target)
      '00000000-0000-0000-0000-000000000011'::uuid,  -- Attacker's user_id
      '22000000-0000-0000-0000-000000000001'::uuid,  -- Tenant B slot
      '2026-09-01',
      'pending'
    );
    RAISE NOTICE 'FAIL: Cross-tenant INSERT succeeded — WITH CHECK clause missing or misconfigured';
  EXCEPTION
    WHEN insufficient_privilege THEN
      RAISE NOTICE 'PASS: Cross-tenant INSERT blocked by RLS WITH CHECK (insufficient_privilege)';
    WHEN others THEN
      RAISE NOTICE 'PASS (other): Cross-tenant INSERT blocked: %', SQLERRM;
  END;
  $$;

ROLLBACK;

-- -----------------------------------------------------------------------------
-- Control: Client in Tenant A can INSERT booking for own tenant
-- Expected: INSERT succeeds (returns INSERT 1)
-- -----------------------------------------------------------------------------
SELECT '=== Control Test: Own-tenant INSERT should succeed ==='::TEXT AS test;

BEGIN;
  SET LOCAL role TO authenticated;
  SET LOCAL request.jwt.claims = '{"sub":"00000000-0000-0000-0000-000000000011","app_metadata":{"tenant_id":"10000000-0000-0000-0000-000000000001","role":"client"}}';

  DO $$
  BEGIN
    INSERT INTO bookings (id, tenant_id, user_id, slot_id, booking_date, status)
    VALUES (
      gen_random_uuid(),
      '10000000-0000-0000-0000-000000000001'::uuid,  -- Own tenant (Tenant A)
      '00000000-0000-0000-0000-000000000011'::uuid,  -- Own user_id
      '12000000-0000-0000-0000-000000000001'::uuid,  -- Tenant A slot
      '2026-09-15',
      'pending'
    );
    RAISE NOTICE 'PASS: Own-tenant INSERT succeeded as expected';
  EXCEPTION
    WHEN others THEN
      RAISE NOTICE 'FAIL: Own-tenant INSERT blocked unexpectedly: %', SQLERRM;
  END;
  $$;

ROLLBACK;

SELECT 'rls_insert_isolation_test.sql complete. Review PASS/FAIL above.'::TEXT AS summary;
```
