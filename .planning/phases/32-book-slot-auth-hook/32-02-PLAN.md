```typescript
// supabase/functions/custom-access-token/index.ts
// Custom Access Token Hook — injeta tenant_id e role em app_metadata
//
// Tipo de hook: Custom Access Token (roda em todo login + refresh)
// Supabase Auto-injects: SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY
//
// SECURITY NOTES:
// - CVE-2025-48757: SEMPRE usar app_metadata (server-set), NUNCA user_metadata
// - Service role key via Deno.env.get() — nunca hardcoded
// - Spread completo de claims existentes para não perder iss/aud/exp/sub
//
// Payload recebido DO Supabase Auth:
//   { user_id: string, claims: object, authentication_method: string }
// Payload retornado PARA Supabase Auth:
//   { claims: { ...original_claims, app_metadata: { tenant_id, role } } }

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.38";

serve(async (req: Request) => {
  // Auth Hooks only accept POST
  if (req.method !== "POST") {
    return new Response(JSON.stringify({ error: "Method not allowed" }), {
      status: 405,
      headers: { "Content-Type": "application/json" },
    });
  }

  // Validate required environment variables before any logic
  const supabaseUrl = Deno.env.get("SUPABASE_URL");
  const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

  if (!supabaseUrl || !serviceRoleKey) {
    console.error("FATAL: SUPABASE_URL or SUPABASE_SERVICE_ROLE_KEY not set");
    return new Response(
      JSON.stringify({ error: "Auth hook misconfigured — missing env vars" }),
      { status: 500, headers: { "Content-Type": "application/json" } }
    );
  }

  try {
    // Parse Auth Hook payload from Supabase Auth server
    const event = await req.json();
    const user_id: string = event.user_id;
    const claims: Record<string, unknown> = event.claims ?? {};

    if (!user_id) {
      console.error("Auth Hook payload missing user_id");
      return new Response(JSON.stringify({ claims }), {
        status: 200,
        headers: { "Content-Type": "application/json" },
      });
    }

    // Initialize Supabase Admin client with service role key
    // This bypasses RLS — needed to query memberships for any user
    const supabaseAdmin = createClient(supabaseUrl, serviceRoleKey, {
      auth: { persistSession: false },
    });

    // Query memberships to get tenant_id + role for this user
    // Uses .maybeSingle() instead of .single() to avoid error when no membership
    const { data: membership, error } = await supabaseAdmin
      .from("memberships")
      .select("tenant_id, role")
      .eq("user_id", user_id)
      .maybeSingle();

    if (error) {
      console.error(`Membership query error for user ${user_id}:`, error.message);
      // Return claims unchanged — RLS will deny access (defense-in-depth)
      return new Response(JSON.stringify({ claims }), {
        status: 200,
        headers: { "Content-Type": "application/json" },
      });
    }

    if (!membership) {
      // User has no membership — not part of any tenant
      // Return claims unchanged; RLS policies will deny access to tenant data
      console.warn(`No membership found for user ${user_id} — returning claims unchanged`);
      return new Response(JSON.stringify({ claims }), {
        status: 200,
        headers: { "Content-Type": "application/json" },
      });
    }

    // Inject tenant_id and role into app_metadata
    // CRITICAL: spread existing claims first to preserve iss, aud, exp, sub, role, etc.
    // CRITICAL: app_metadata only (NOT user_metadata) — CVE-2025-48757
    const modifiedClaims = {
      ...claims,
      app_metadata: {
        ...((claims.app_metadata as Record<string, unknown>) ?? {}),
        tenant_id: membership.tenant_id,
        role: membership.role,
      },
    };

    console.log(
      `Auth Hook: injected tenant_id=${membership.tenant_id} role=${membership.role} for user ${user_id}`
    );

    return new Response(JSON.stringify({ claims: modifiedClaims }), {
      status: 200,
      headers: { "Content-Type": "application/json" },
    });
  } catch (err) {
    console.error("Auth Hook unexpected error:", err);
    // Return 500 — Supabase Auth will fail login (safer than issuing unmodified token)
    return new Response(
      JSON.stringify({ error: "Internal server error in auth hook" }),
      { status: 500, headers: { "Content-Type": "application/json" } }
    );
  }
});
```

Pontos críticos na implementação:
1. Usar `.maybeSingle()` (não `.single()`) para não lançar erro quando usuário não tem membership
2. Spread `...claims` ANTES de adicionar app_metadata — preserva iss, aud, exp, sub, role
3. Spread `...claims.app_metadata` ao adicionar campos — preserva qualquer app_metadata preexistente
4. Validar env vars antes de qualquer lógica — falha rápida com log claro
5. Retornar 200 com claims não modificados quando não há membership (não 4xx — 4xx quebraria o login)
6. Retornar 500 em erro inesperado (Supabase Auth negará login — mais seguro que emitir token sem tenant_id)

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Supabase Auth → Edge Function | POST request com payload { user_id, claims } — origem é Supabase Auth server, não cliente externo; mas payload deve ser validado |
| Edge Function → public.memberships | Service role key bypassa RLS — acesso irrestrito ao banco; key deve ser somente via env var |
| JWT → RLS policies | app_metadata injetado pelo hook é server-set e imutável pelo cliente — base de confiança para políticas da Phase 31 |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-32-02-01 | Spoofing | Auth Hook payload user_id | accept | Payload vem diretamente do Supabase Auth server (não do cliente); Supabase valida origem internamente. Spike não implementa Standard Webhooks signature verification (deferred para produção). |
| T-32-02-02 | InfoDisc | SUPABASE_SERVICE_ROLE_KEY em logs | mitigate | Código não loga a key — apenas loga mensagem de erro "missing env vars" sem expor valor. Console.log usa apenas user_id e tenant_id (não secrets). |
| T-32-02-03 | EoP | user_metadata no JWT | mitigate | `grep -c 'user_metadata' supabase/functions/custom-access-token/index.ts` = 0. CVE-2025-48757: zero ocorrências em código executável. app_metadata APENAS. |
| T-32-02-04 | Tampering | Auth Hook retorna claims incompletos | mitigate | Spread `...claims` antes de modificar app_metadata — garante que iss, aud, exp, sub, role são preservados. Pitfall 5 da RESEARCH.md explicitamente coberto. |
| T-32-02-05 | DoS | Auth Hook timeout bloqueia login | mitigate | .maybeSingle() tem timeout padrão Supabase (~5s). Erros retornam 200 com claims originais (não 5xx) para não bloquear login quando membership ausente. Erro inesperado retorna 500 (login falha — mais seguro). |
| T-32-02-06 | InfoDisc | Service role key hardcoded em index.ts | mitigate | Validação explícita `if (!supabaseUrl \|\| !serviceRoleKey)` com FATAL log. Key NUNCA aparece em código — somente `Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')`. |
</threat_model>
